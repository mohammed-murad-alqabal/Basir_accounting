الحصول على إحصائيات الفواتير

يحسب ويسترجع إحصائيات شاملة عن جميع الفواتير.

Returns: كائن InvoiceStatistics يحتوي على الإحصائيات

مثال:

final stats = await repository.getInvoiceStatistics();
print('إجمالي الإيرادات: ${stats.totalRevenue}');

Implementation

@override
Future<InvoiceStatistics> getInvoiceStatistics() async {
  try {
    final allInvoices = await getAllInvoices();

    final paidInvoices =
        allInvoices.where((invoice) => invoice.status == 'paid').toList();

    final overdueInvoices =
        allInvoices.where((invoice) => invoice.status == 'overdue').toList();

    final totalRevenue = allInvoices.fold<double>(
      0,
      (sum, invoice) => sum + invoice.grandTotal,
    );
    final paidRevenue = paidInvoices.fold<double>(
      0,
      (sum, invoice) => sum + invoice.grandTotal,
    );

    return InvoiceStatistics(
      totalInvoices: allInvoices.length,
      paidInvoices: paidInvoices.length,
      overdueInvoices: overdueInvoices.length,
      totalRevenue: totalRevenue,
      paidRevenue: paidRevenue,
    );
  } catch (e) {
    throw Exception('خطأ في حساب إحصائيات الفواتير: $e');
  }
}
