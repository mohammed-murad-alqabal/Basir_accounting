```dart
    // implementation
  }

// مثال محسن
class DataService {
  /// يجلب البيانات من المصدر المحلي مع معالجة شاملة للأخطاء
  ///
  /// Returns: قائمة البيانات أو قائمة فارغة في حالة الخطأ
  /// Throws: [DataException] في حالة فشل الاتصال
  Future<List<Data>> getData() async {
    try {
      final data = await _localRepository.fetchAll();
      return data ?? [];
    } catch (e) {
      throw DataException('فشل في جلب البيانات: $e');
    }
  }
}
```
